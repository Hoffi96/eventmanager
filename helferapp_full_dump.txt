===== Services/HtmlSanitizer.cs =====
0001: using System.Text.RegularExpressions;
0002: 
0003: namespace HelferApp.Services;
0004: 
0005: /// <summary>
0006: /// Einfache, regelbasierte Bereinigung von HTML aus dem WYSIWYG-Editor (Quill).
0007: /// Nur Admins können aktuell Task-/Event-Beschreibungen speichern, trotzdem
0008: /// als Verteidigungslinie: entfernt gefährliche Tags, Event-Handler-Attribute
0009: /// und javascript:-Links. Kein Ersatz für eine vollwertige Sanitizer-Bibliothek,
0010: /// aber ohne zusätzliche NuGet-Abhängigkeit nutzbar.
0011: /// </summary>
0012: public static class HtmlSanitizer
0013: {
0014:     private static readonly string[] ForbiddenTags =
0015:     {
0016:         "script", "iframe", "object", "embed", "form", "input", "link", "meta", "style", "svg"
0017:     };
0018: 
0019:     public static string Sanitize(string? html)
0020:     {
0021:         if (string.IsNullOrWhiteSpace(html))
0022:         {
0023:             return "";
0024:         }
0025: 
0026:         var result = html;
0027: 
0028:         foreach (var tag in ForbiddenTags)
0029:         {
0030:             result = Regex.Replace(result, $"<{tag}[^>]*>.*?</{tag}>", "", RegexOptions.IgnoreCase | RegexOptions.Singleline);
0031:             result = Regex.Replace(result, $"<{tag}[^>]*/?>", "", RegexOptions.IgnoreCase);
0032:         }
0033: 
0034:         // Event-Handler-Attribute (onclick=, onerror= ...) entfernen
0035:         result = Regex.Replace(result, @"\s+on\w+\s*=\s*(""[^""]*""|'[^']*'|[^\s>]+)", "", RegexOptions.IgnoreCase);
0036: 
0037:         // javascript:-Links entschärfen
0038:         result = Regex.Replace(result, @"(href|src)\s*=\s*(""|')\s*javascript:[^""']*(""|')", "$1=\"#\"", RegexOptions.IgnoreCase);
0039: 
0040:         result = result.Trim();
0041: 
0042:         // Quill liefert bei leerem Editor "<p><br></p>" statt eines leeren Strings.
0043:         if (result == "<p><br></p>")
0044:         {
0045:             return "";
0046:         }
0047: 
0048:         return result;
0049:     }
0050: }

===== Services/PasswordHelper.cs =====
0001: using System.Security.Cryptography;
0002: 
0003: namespace HelferApp.Services;
0004: 
0005: /// <summary>
0006: /// Einfaches, sicheres Passwort-Hashing per PBKDF2 (SHA-256), ohne Abhängigkeit
0007: /// von Microsoft.AspNetCore.Identity. Format: "{salt-base64}:{hash-base64}".
0008: /// </summary>
0009: public static class PasswordHelper
0010: {
0011:     private const int SaltSize = 16;
0012:     private const int HashSize = 32;
0013:     private const int Iterations = 100_000;
0014: 
0015:     public static string Hash(string password)
0016:     {
0017:         var salt = RandomNumberGenerator.GetBytes(SaltSize);
0018:         var hash = Rfc2898DeriveBytes.Pbkdf2(password, salt, Iterations, HashAlgorithmName.SHA256, HashSize);
0019:         return $"{Convert.ToBase64String(salt)}:{Convert.ToBase64String(hash)}";
0020:     }
0021: 
0022:     public static bool Verify(string password, string stored)
0023:     {
0024:         var parts = stored.Split(':');
0025:         if (parts.Length != 2)
0026:         {
0027:             return false;
0028:         }
0029: 
0030:         var salt = Convert.FromBase64String(parts[0]);
0031:         var expected = Convert.FromBase64String(parts[1]);
0032:         var actual = Rfc2898DeriveBytes.Pbkdf2(password, salt, Iterations, HashAlgorithmName.SHA256, HashSize);
0033: 
0034:         return CryptographicOperations.FixedTimeEquals(expected, actual);
0035:     }
0036: }
===== Services/SmtpEmailService.cs =====
0001: using System.Net;
0002: using System.Net.Mail;
0003: using Microsoft.Extensions.Logging;
0004: using Microsoft.Extensions.Options;
0005: 
0006: namespace HelferApp.Services;
0007: 
0008: /// <summary>
0009: /// Versendet E-Mails per SMTP, sofern in appsettings.json unter "Email"
0010: /// konfiguriert (Enabled = true + SmtpHost gesetzt). Andernfalls wird die
0011: /// Mail nur geloggt, damit die App auch ohne Mail-Setup lauffähig bleibt.
0012: /// </summary>
0013: public class SmtpEmailService : IEmailService
0014: {
0015:     private readonly EmailOptions _options;
0016:     private readonly ILogger<SmtpEmailService> _logger;
0017: 
0018:     public SmtpEmailService(IOptions<EmailOptions> options, ILogger<SmtpEmailService> logger)
0019:     {
0020:         _options = options.Value;
0021:         _logger = logger;
0022:     }
0023: 
0024:     public async Task SendAsync(string toAddress, string subject, string body)
0025:     {
0026:         if (string.IsNullOrWhiteSpace(toAddress))
0027:         {
0028:             return;
0029:         }
0030: 
0031:         if (!_options.Enabled || string.IsNullOrWhiteSpace(_options.SmtpHost))
0032:         {
0033:             _logger.LogInformation(
0034:                 "E-Mail nicht versendet (SMTP deaktiviert/nicht konfiguriert) an {To}: {Subject}\n{Body}",
0035:                 toAddress, subject, body);
0036:             return;
0037:         }
0038: 
0039:         using var message = new MailMessage
0040:         {
0041:             From = new MailAddress(_options.FromAddress, _options.FromName),
0042:             Subject = subject,
0043:             Body = body,
0044:             IsBodyHtml = false
0045:         };
0046:         message.To.Add(toAddress);
0047: 
0048:         using var client = new SmtpClient(_options.SmtpHost, _options.SmtpPort)
0049:         {
0050:             EnableSsl = _options.EnableSsl,
0051:             Credentials = string.IsNullOrWhiteSpace(_options.SmtpUser)
0052:                 ? CredentialCache.DefaultNetworkCredentials
0053:                 : new NetworkCredential(_options.SmtpUser, _options.SmtpPassword)
0054:         };
0055: 
0056:         try
0057:         {
0058:             await client.SendMailAsync(message);
0059:         }
0060:         catch (Exception ex)
0061:         {
0062:             _logger.LogError(ex, "E-Mail-Versand an {To} fehlgeschlagen", toAddress);
0063:         }
0064:     }
0065: }
